    function binaryRule(name: string, next: string, operators: string[]) {
        return createRule(name,
        seq(
            rule(next),
            zeroOrMore(seq(choice(...operators.map(operator => operator === '*' ? token('*', '*') : token(operator))), rule(next))),
        ),
        {
            build: (r: Result) => {
                const parts = r.getSequenceResult()!;
                let result = getExpr(parts[0]);
                for (const item of parts[1].getRepeatResult() ?? []) {
                    const pair = item.getSequenceResult()!;
                    const operator = pair[0].getChoiceResult()?.getTokenData()?.text ?? '';
                    const right = getExpr(pair[1]);
                    result = AST.make.expr.binary(
                        { start: result.span.start, end: right.span.end },
                        result,
                        operator,
                        right,
                    );
                }
                return mkC(name, result, r);
            },
        },
        );
    }
