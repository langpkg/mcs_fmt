    export const stmtRules: Rules = [

        createRule('LetStmt',
            seq(
                optional(rule('Visibility')),
                token('let'),
                optional(token('mut')),
                token('ident'),
                optional(seq(token(':'), rule('Type'))),
                token('='),
                rule('Expr'),
                optional(token(';')),
            ),
            {
                build: (r: Result) => {
                    // extract
                    const parts = r.getSequenceResult()!;
                    const visibility = parts[0].isOptionalPassed()
                    ? parts[0].getOptionalResult()!.getCustomData() as AST.VisibilityModifier
                    : null;
                    const name = parts[3].getTokenData()!.text;
                    const type = parts[4].isOptionalPassed()
                    ? getType(parts[4].getOptionalResult()!)
                    : null;
                    const value = getExpr(parts[6]);

                    if (!parts[7].isOptionalPassed()) {
                        return r.withError(mkE('Expected let terminator ";"', 'MISSING_SEMICOLON', sp(r)));
                    }

                    // generate
                    const stmt = AST.make.stmt.let( sp(r), name, parts[2].isOptionalPassed(), type, value, visibility, );
                    return mkC('LetStmt', stmt);
                },
                errors: [errorFn(0, 'Expected let statement')],
            }
        ),

        createRule('FnStmt',
        conditional(seq(
            optional(rule('Visibility')),
            optional(token('extern')),
            token('fn'),
            token('ident'),
            token('('),
            optional(zeroOrMore(seq(rule('Param'), optional(token(','))))),
            token(')'),
            optional(seq(token(':'), rule('Type'))),
            optional(rule('BlockStmt')),
            optional(token(';')),
        ), () => true),
        {
            build: (r: Result) => {
                // extract
                const sequence = r.isCustom('conditional')
                ? r.getCustomData() as Result
                : r;
                const parts = sequence.getSequenceResult()!;
                const visibility = parts[0].isOptionalPassed()
                ? parts[0].getOptionalResult()!.getCustomData() as AST.VisibilityModifier
                : null;
                const extern: AST.ExternModifier | null = parts[1].isOptionalPassed()
                ? {
                    kind: 'ExternModifier',
                    span: parts[1].getOptionalResult()!.span,
                    mode: 'extern',
                }
                : null;
                const params = parts[5].isOptionalPassed()
                ? (parts[5].getOptionalResult()!.getRepeatResult() ?? []).map(result =>
                result.getSequenceResult()![0].getCustomData() as AST.Param)
                : [];
                const returnType = parts[7].isOptionalPassed()
                ? getType(parts[7].getOptionalResult()!)
                : null;
                const body = parts[8].isOptionalPassed()
                ? parts[8].getOptionalResult()!.getCustomData() as AST.BlockStmt
                : null;
                const hasTerminator = parts[9].isOptionalPassed();

                if (extern && body) {
                    return r.withError(mkE("Extern function declaration cannot have a body", 'INVALID_EXTERN_BODY', sp(r)));
                }

                if (extern && !body && !hasTerminator) {
                    return r.withError(mkE("Extern function declaration requires ';'", 'MISSING_SEMICOLON', sp(r)));
                }

                if (!extern && !body) {
                    return r.withError(mkE("Function body is required '{ ... }'", 'MISSING_FUNCTION_BODY', sp(r)));
                }

                const stmt = AST.make.stmt.fn(sp(r), parts[3].getTokenData()!.text, visibility, extern, params, returnType, body);
                return mkC('FnStmt', stmt);
            },

            errors: [errorFn(0, 'Expected function statement')],
        }
        ),

    ];
